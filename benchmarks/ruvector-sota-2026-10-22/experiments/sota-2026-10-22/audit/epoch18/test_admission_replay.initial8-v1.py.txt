def put(root, path, value):
    file = root/path; file.parent.mkdir(parents=True, exist_ok=True)
    file.write_text(json.dumps(value, sort_keys=True))
    return S.sha(file)

class AdmissionReplayFixtures(unittest.TestCase):
    def test_final_capacity_can_differ_only_after_source_admission(self):
        with tempfile.TemporaryDirectory() as td:
            root=Path(td); final,_=admission_fixture(root)
            self.assertTrue(S.verify_operational_admission(final,root))

    def test_operational_template_cannot_masquerade_as_final(self):
        for key, bad in [('measurement_plan',False),('operational_prelabel_only',True)]:
            with tempfile.TemporaryDirectory() as td:
                root=Path(td); final,_=admission_fixture(root); final[key]=bad
                with self.subTest(key=key),self.assertRaises(AssertionError):
                    S.verify_operational_admission(final,root)
        with tempfile.TemporaryDirectory() as td:
            root=Path(td); final,_=admission_fixture(root)
            final['runtime_admission']['capacity_pending_label_free_inputs']=True
            with self.assertRaises(AssertionError):S.verify_operational_admission(final,root)

    def test_model_budget_project_lease_or_formula_drift_fails(self):
        for key,bad in [('candidate_manifest_sha256','c'*64),('actual_project','/different'),
                        ('lease_expires_unix',123457),('new_study_native_attempt_cap',3),
